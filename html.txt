    const icon = name => `<svg class="icon"><use href="#${name}"/></svg>`;
    const escapeHTML = value => String(value).replace(/[&<>"']/g, char => ({ '&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#39;' }[char]));

    function renderPlans(query = '') {
      const filtered = plans.map((plan, index) => ({ plan, index })).filter(({ plan }) => plan.name.toLowerCase().includes(query.toLowerCase()) || plan.region.toLowerCase().includes(query.toLowerCase()));
      const row = ({ plan, index }, full) => `<tr>
        <td><div class="plan-cell"><span class="plan-icon">${icon('i-server')}</span><strong>${escapeHTML(plan.name)}</strong></div></td>
        <td>${escapeHTML(plan.region)}</td><td>${escapeHTML(plan.backup)}</td>
        ${full ? `<td>${escapeHTML(plan.rpo)} / ${escapeHTML(plan.rto)}</td>` : ''}
        <td><span class="badge ${plan.status === 'Protected' ? 'green' : plan.status === 'Testing' ? 'blue' : 'amber'}">${plan.status === 'Protected' ? '<span class="status-dot"></span>' : ''}${plan.status}</span></td>
        <td><button class="row-action" data-plan="${index}" aria-label="View ${escapeHTML(plan.name)} details">${icon('i-more')}</button></td>
      </tr>`;
      $('overviewPlanRows').innerHTML = filtered.slice(0, 4).map(item => row(item, false)).join('') || '<tr><td colspan="5" class="empty-state">No matching plans found.</td></tr>';
      $('allPlanRows').innerHTML = filtered.map(item => row(item, true)).join('') || '<tr><td colspan="6" class="empty-state">No matching plans found.</td></tr>';
    }

    function renderActivities() {
      $('activityList').innerHTML = activities.slice(0, 4).map(activity => `<div class="activity-entry"><span class="activity-icon ${activity.type}">${icon(activity.icon)}</span><div class="activity-text"><strong>${escapeHTML(activity.title)}</strong><span>${escapeHTML(activity.detail)}</span></div></div>`).join('');
    }

    function showView(view) {
      currentView = view;
      document.querySelectorAll('.nav-link').forEach(link => link.classList.toggle('active', link.dataset.view === view));
      document.querySelectorAll('#overview, .subview').forEach(section => section.classList.toggle('active', section.id === view));
      const names = { overview: 'Overview', plans: 'Recovery Plans', backups: 'Backups', testing: 'Failover Testing', reports: 'Reports', settings: 'Settings' };
      const subtitles = {
        plans: 'Keep every critical workload protected and ready to recover.',
        backups: 'Track the health and history of your backup operations.',
        testing: 'Verify your business can bounce back, before it needs to.',
        reports: 'The insights you need to stay resilient and compliant.',
        settings: 'Manage your workspace and protection preferences.'
      };
      $('breadcrumb').textContent = names[view];
      $('pageTitle').textContent = view === 'overview' ? 'Good morning, Alex 👋' : names[view];
      $('pageSubtitle').textContent = view === 'overview' ? "Here's what's happening with your disaster recovery today." : subtitles[view];
      $('sidebar').classList.remove('open');
      $('mobileOverlay').classList.remove('open');
      $('notificationsPopover').classList.remove('open');
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function showToast(message) {
      $('toastText').textContent = message;
      $('toast').classList.add('show');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => $('toast').classList.remove('show'), 3500);
    }

    function openModal(title, subtitle, content) {
      $('modalTitle').textContent = title;
      $('modalSubtitle').textContent = subtitle;
      $('modalBody').innerHTML = content;
      $('modalBackdrop').classList.add('open');
      document.body.style.overflow = 'hidden';
      const firstField = $('modalBody').querySelector('input, select');
      if (firstField) setTimeout(() => firstField.focus(), 80);
    }
    function closeModal() {
      $('modalBackdrop').classList.remove('open');
      document.body.style.overflow = '';
    }
    const modalActions = (action, label) => `<div class="modal-actions"><button class="btn btn-secondary" data-action="close">Cancel</button><button class="btn btn-primary" data-action="${action}">${label}</button></div>`;

    function createPlanModal() {
      openModal('Create recovery plan', 'Set up protection for a critical workload.', `
        <form id="createPlanForm">
          <div class="field"><label for="planName">Plan name</label><input id="planName" name="name" placeholder="e.g. Billing Service" maxlength="60" required></div>
          <div class="field"><label for="planRegion">Backup region</label><select id="planRegion" name="region"><option>US East (N. Virginia)</option><option>US West (Oregon)</option><option>EU (Frankfurt)</option><option>Asia Pacific (Singapore)</option></select></div>
          <div class="form-two"><div class="field"><label for="planRpo">Recovery point (RPO)</label><select id="planRpo" name="rpo"><option>15 min</option><option>30 min</option><option>1 hour</option><option>4 hours</option></select></div><div class="field"><label for="planRto">Recovery time (RTO)</label><select id="planRto" name="rto"><option>30 min</option><option>1 hour</option><option>2 hours</option><option>4 hours</option></select></div></div>
          <div class="modal-actions"><button type="button" class="btn btn-secondary" data-action="close">Cancel</button><button type="submit" class="btn btn-primary">Create plan</button></div>
        </form>`);
    }

    function testModal() {
      openModal('Run failover test', 'Safely verify a recovery plan without affecting production.', `
        <div class="field"><label for="testPlan">Choose recovery plan</label><select id="testPlan">${plans.map((p, i) => `<option value="${i}">${escapeHTML(p.name)}</option>`).join('')}</select></div>
        <div class="readiness-note">${icon('i-shield')} This is a non-disruptive simulation. Production stays online.</div>
        ${modalActions('confirm-test', 'Start test')}`);
    }

    function backupModal() {
      openModal('Run an on-demand backup', 'Create a new recovery point for a protected workload.', `
        <div class="field"><label for="backupPlan">Choose recovery plan</label><select id="backupPlan">${plans.map((p, i) => `<option value="${i}">${escapeHTML(p.name)}</option>`).join('')}</select></div>
        <div class="readiness-note">${icon('i-lock')} Your backup will be encrypted and replicated securely.</div>
        ${modalActions('confirm-backup', 'Run backup')}`);
    }

    function planDetail(index) {
      const plan = plans[index];
      if (!plan) return;
      openModal(plan.name, 'Recovery plan details', `
        <div class="detail-line"><span>Status</span><strong>${escapeHTML(plan.status)}</strong></div>
        <div class="detail-line"><span>Backup region</span><strong>${escapeHTML(plan.region)}</strong></div>
        <div class="detail-line"><span>Last backup</span><strong>${escapeHTML(plan.backup)}</strong></div>
        <div class="detail-line"><span>Recovery point objective</span><strong>${escapeHTML(plan.rpo)}</strong></div>
        <div class="detail-line"><span>Recovery time objective</span><strong>${escapeHTML(plan.rto)}</strong></div>
        <div class="modal-actions"><button class="btn btn-secondary" data-action="close">Close</button><button class="btn btn-primary" data-action="test-plan" data-index="${index}">Test this plan</button></div>`);
    }

    function addActivity(title, detail, type, activityIcon) {
      activities.unshift({ title, detail, type, icon: activityIcon });
      renderActivities();
    }

    function downloadReport(type) {
      const date = new Date().toISOString().slice(0, 10);
      const reports = {
        readiness: `CloudShield Recovery Readiness Report\nGenerated: ${date}\n\nOverall readiness: 94%\nProtected workloads: 48\nRecovery plans: ${totalPlans}\nActive plans: ${activePlans}\n\nPlan,Region,Status,RPO,RTO\n${plans.map(p => `"${p.name}","${p.region}",${p.status},${p.rpo},${p.rto}`).join('\n')}`,
        backups: `CloudShield Backup Activity Report\nGenerated: ${date}\n\nBackup success rate: 99.8%\nProtected data: 12.4 TB\nLast successful backup: ${$('lastBackupDisplay').textContent}\n\nPlan,Last Backup,Status\n${plans.map(p => `"${p.name}",${p.backup},${p.status}`).join('\n')}`,
        tests: `CloudShield Failover Test Report\nGenerated: ${date}\n\nTests this month: ${$('testCount').textContent}\nLast test result: Passed\nAverage recovery time: 18 minutes\n\nRecent tests:\nProduction Database: Passed, 14 min\nCustomer Portal: Passed, 17 min\nAnalytics Platform: Passed, 22 min`
      };
      const blob = new Blob([reports[type]], { type: 'text/plain;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      link.download = `cloudshield-${type}-report-${date}.txt`;
      document.body.appendChild(link);
      link.click();
      link.remove();
      setTimeout(() => URL.revokeObjectURL(url), 1000);
      showToast('Report downloaded successfully.');
    }

    const chartSets = {
      '7': { a: [78,83,80,88,85,91,89,94,90,96,92,97,94], b: [68,72,70,76,74,81,78,82,80,87,84,89,86], labels: ['Mon','Tue','Wed','Thu','Fri','Sat','Sun'] },
      '30': { a: [76,79,77,83,80,87,85,88,86,91,89,94,96], b: [65,68,72,69,75,72,78,77,80,82,79,85,88], labels: ['Apr 1','Apr 5','Apr 10','Apr 15','Apr 20','Apr 25','Apr 30'] },
      '90': { a: [73,77,75,80,78,83,82,87,85,89,90,93,96], b: [62,66,64,70,72,68,75,77,76,82,80,85,88], labels: ['Feb 1','Feb 15','Mar 1','Mar 15','Apr 1','Apr 15','Apr 30'] }
    };
    function drawChart(period) {
      const data = chartSets[period];
      const points = values => values.map((value, i) => `${i * 50},${150 - value * 1.5}`).join(' ');
      const path = values => 'M' + points(values).replaceAll(' ', ' L');
      $('primaryPath').setAttribute('d', path(data.a));
      $('secondaryPath').setAttribute('d', path(data.b));
      $('areaPath').setAttribute('d', `${path(data.a)} L600,150 L0,150 Z`);
      $('chartLabels').innerHTML = data.labels.map(label => `<span>${label}</span>`).join('');
    }

    document.addEventListener('click', event => {
      const viewButton = event.target.closest('[data-view]');
      if (viewButton) { event.preventDefault(); showView(viewButton.dataset.view); return; }
      const planButton = event.target.closest('[data-plan]');
      if (planButton) { planDetail(Number(planButton.dataset.plan)); return; }
      const reportButton = event.target.closest('[data-report]');
      if (reportButton) { downloadReport(reportButton.dataset.report); return; }
      const actionButton = event.target.closest('[data-action]');
      if (!actionButton) return;
      switch (actionButton.dataset.action) {
        case 'create': createPlanModal(); break;
        case 'backup': backupModal(); break;
        case 'test': testModal(); break;
        case 'close': closeModal(); break;
        case 'test-plan': closeModal(); testModal(); $('testPlan').value = actionButton.dataset.index; break;
        case 'confirm-backup': {
          const plan = plans[Number($('backupPlan').value)];
          plan.backup = 'Just now';
          $('lastBackupDisplay').textContent = 'Just now';
          $('backupJobs').insertAdjacentHTML('afterbegin', `<div class="info-row"><div class="info-row-left">${icon('i-cloud')}<div><strong>${escapeHTML(plan.name)}</strong><small>Completed just now · On-demand backup</small></div></div><span class="badge green">Successful</span></div>`);
          addActivity('On-demand backup completed', `${plan.name} · Just now`, 'green', 'i-check');
          renderPlans($('globalSearch').value);
          closeModal();
          showToast(`Backup completed for ${plan.name}.`);
          break;
        }
        case 'confirm-test': {
          const plan = plans[Number($('testPlan').value)];
          $('testCount').textContent = Number($('testCount').textContent) + 1;
          $('testHistory').insertAdjacentHTML('afterbegin', `<div class="info-row"><div class="info-row-left">${icon('i-shield')}<div><strong>${escapeHTML(plan.name)}</strong><small>Just now · Recovery checks succeeded</small></div></div><span class="badge green">Passed</span></div>`);
          addActivity('Failover test passed', `${plan.name} · Just now`, 'blue', 'i-shield');
          closeModal();
          showToast(`Failover test passed for ${plan.name}.`);
          break;
        }
      }
    });

    document.addEventListener('submit', event => {
      if (event.target.id !== 'createPlanForm') return;
      event.preventDefault();
      const form = new FormData(event.target);
      const name = String(form.get('name')).trim();
      if (!name) return;
      plans.unshift({ name, region: form.get('region'), backup: 'Pending', status: 'Protected', rpo: form.get('rpo'), rto: form.get('rto') });
      totalPlans++;
      activePlans++;
      $('planCount').textContent = totalPlans;
      $('navPlanCount').textContent = totalPlans;
      $('activePlanCount').textContent = activePlans;
      $('globalSearch').value = '';
      renderPlans();
      addActivity('New recovery plan created', `${name} · Just now`, 'green', 'i-layers');
      closeModal();
      showView('plans');
      showToast(`${name} is now protected.`);
    });

    $('newPlanButton').addEventListener('click', createPlanModal);
    $('runBackupButton').addEventListener('click', backupModal);
    $('bannerTestButton').addEventListener('click', testModal);
    $('closeModal').addEventListener('click', closeModal);
    $('modalBackdrop').addEventListener('click', event => { if (event.target === $('modalBackdrop')) closeModal(); });
    document.addEventListener('keydown', event => { if (event.key === 'Escape') { closeModal(); $('notificationsPopover').classList.remove('open'); } });
    $('chartPeriod').addEventListener('change', event => drawChart(event.target.value));
    $('globalSearch').addEventListener('input', event => {
      renderPlans(event.target.value);
      if (event.target.value.trim()) showView('plans');
    });
    $('menuButton').addEventListener('click', () => { $('sidebar').classList.add('open'); $('mobileOverlay').classList.add('open'); });
    $('mobileOverlay').addEventListener('click', () => { $('sidebar').classList.remove('open'); $('mobileOverlay').classList.remove('open'); });
    $('notificationsButton').addEventListener('click', () => {
      const open = $('notificationsPopover').classList.toggle('open');
      $('notificationsButton').setAttribute('aria-expanded', open);
    });
    $('helpButton').addEventListener('click', () => showToast('Help center: contact support@cloudshield.example for assistance.'));
    document.querySelectorAll('.switch').forEach(button => button.addEventListener('click', () => {
      button.classList.toggle('on');
      const enabled = button.classList.contains('on');
      button.setAttribute('aria-checked', String(enabled));
      showToast(`${button.getAttribute('aria-label')} ${enabled ? 'enabled' : 'disabled'}.`);
    }));

    renderPlans();
    renderActivities();
    drawChart('7');
  </script>
